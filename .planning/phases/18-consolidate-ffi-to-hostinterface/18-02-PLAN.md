//! These functions populate the HostInterface struct and are called

use polyplug_abi::{HostInterface, AbiError, ...};
use crate::runtime::Runtime;

/// Extract Runtime reference from HostInterface runtime field.
fn runtime_from_host(host: *const HostInterface) -> Option<&Runtime> {
    if host.is_null() { return None; }

    unsafe { (host as *const HostInterface).read().runtime }
        .map(|ptr| unsafe { &*(ptr as *const Runtime) })

Implementation approach:
```rust
// In ffi.rs
#[unsafe(no_mangle)]
pub unsafe extern "C" fn polyplug_runtime_create() -> *const HostInterface {
    std::panic::catch_unwind(|| {
        match Runtime::builder().build() {
            Ok(rt) => {
                // Runtime now owns itself via Box::leak for 'static lifetime
                // host_abi field contains the HostInterface pointer
                rt.host_abi as *const HostInterface
            },
            Err(_) => core::ptr::null(),
        }
    })
    .unwrap_or(core::ptr::null())
}
```

Runtime::build() needs to:
1. Create Runtime instance
2. Populate HostInterface with function pointers from host_interface_impl module
3. Box::leak both Runtime and HostInterface for 'static lifetime
4. Return HostInterface pointer
