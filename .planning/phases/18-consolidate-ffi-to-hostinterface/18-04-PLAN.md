```lua
local Runtime = {}
Runtime.__index = Runtime

function Runtime:new()
    local host = lib.polyplug_runtime_create()
    if host == nil then
        error("polyplug_runtime_create returned null")
    end
    local obj = {
        _host = host,  -- HostInterface* pointer
        _host_struct = host[0],  -- Dereference to get struct
    }
    setmetatable(obj, Runtime)
    return obj
end

function Runtime:load_bundle(path)
    local path_bytes = path .. "\0"
    local path_len = #path_bytes - 1
    -- Call through HostInterface.load_bundle field
    local func_ptr = self._host_struct.load_bundle
    local func = ffi.cast("uint32_t(*)(const HostInterface*, const uint8_t*, size_t)", func_ptr)
    local result = func(self._host, path_bytes, path_len)
    if result ~= 0 then
        error("load_bundle failed: " .. self:_get_last_error())
    end
end

function Runtime:find_guest_contract(contract_id, min_version)
    local func_ptr = self._host_struct.find_guest_contract
    local func = ffi.cast("uint64_t(*)(const HostInterface*, uint64_t, uint32_t)", func_ptr)
    return func(self._host, contract_id, min_version)
end

-- Helper to get last error through HostInterface
function Runtime:_get_last_error()
    local len_func = ffi.cast("size_t(*)(const HostInterface*)", self._host_struct.get_error_len)
    local len = len_func(self._host)
    if len == 0 then return "" end
    local buf = ffi.new("uint8_t[?]", len)
    local err_func = ffi.cast("size_t(*)(const HostInterface*, uint8_t*, size_t)", self._host_struct.get_last_error)
    err_func(self._host, buf, len)
    return ffi.string(buf, len)
end

function Runtime:destroy()
    if self._host ~= nil then
        lib.polyplug_runtime_destroy(self._host)
        self._host = nil
    end
end
```

```cpp
        if (host_ != nullptr) {
            polyplug_runtime_destroy(host_);
            host_ = nullptr;

    void load_bundle(std::string_view path) {
        ensure_host();
        // Call through HostInterface.load_bundle field
        auto func = reinterpret_cast<uint32_t(*)(const HostInterface*, const uint8_t*, size_t)>(host_->load_bundle);
        uint32_t result = func(host_, reinterpret_cast<const uint8_t*>(path.data()), path.size());
        if (result != 0) {
            throw std::runtime_error(get_last_error());
        }
    }
    
    uint64_t find_guest_contract(uint64_t contract_id, uint32_t min_version) {
        ensure_host();
        auto func = reinterpret_cast<uint64_t(*)(const HostInterface*, uint64_t, uint32_t)>(host_->find_guest_contract);
        return func(host_, contract_id, min_version);
    }
    
    std::string get_last_error() {
        ensure_host();
        auto len_func = reinterpret_cast<size_t(*)(const HostInterface*)>(host_->get_error_len);
        size_t len = len_func(host_);
        if (len == 0) return "";
        std::vector<char> buf(len);
        auto err_func = reinterpret_cast<size_t(*)(const HostInterface*, uint8_t*, size_t)>(host_->get_last_error);
```
