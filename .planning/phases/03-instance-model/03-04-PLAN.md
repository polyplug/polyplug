<objective>
Generate RAII instance wrappers for host-side contract callers that manage instance lifecycle via create_instance/destroy_instance.

Purpose: Host applications use generated wrappers to call plugin methods. Wrappers must create instances on construction, destroy on drop, and pass instance to dispatch calls. This ensures hot-reload safety (instances destroyed before interface swap).

Output: Rust generator produces host callers with instance-based RAII pattern instead of PluginGuard.
</objective>

    interface: *const GuestContractInterface,
    instance: GuestContractInstance,
    rt_ctx: *mut core::ffi::c_void,
}

impl TestAddContract {
    /// Create a new instance wrapper.
    /// Calls create_instance on construction.
    pub fn new(handle: PluginHandle, rt_ctx: *mut core::ffi::c_void) -> Option<Self> {
        let interface: *const GuestContractInterface = unsafe {
            polyplug_abi::ffi::polyplug_runtime_resolve_plugin(
                rt_ctx as *const OpaqueRuntime,
                handle.pack(),
            )
        };
        if interface.is_null() {
            return None;
        }
        let instance: GuestContractInstance = unsafe {
            ((*interface).create_instance)(rt_ctx, core::ptr::null())
        };
        if instance.data.is_null() {
            return None;
        }
        Some(TestAddContract { interface, instance, rt_ctx })
    }
    
    /// Check if instance is valid (non-null).
    pub fn is_valid(&self) -> bool {
        !self.instance.data.is_null()
    }
    
    /// Destroy current instance and create new one.
    pub fn reset(&mut self) {
        unsafe {
            ((*self.interface).destroy_instance)(self.rt_ctx, self.instance);
        }
        self.instance = unsafe {
            ((*self.interface).create_instance)(self.rt_ctx, core::ptr::null())
        };
    }
    
    pub fn add(&self, a: i32, b: i32) -> Result<i32, ContractError> {
        // dispatch with instance as first arg...
    }
}

impl Drop for TestAddContract {
    fn drop(&mut self) {
        if !self.instance.data.is_null() {
            unsafe {
                ((*self.interface).destroy_instance)(self.rt_ctx, self.instance);
            }
        }
    }
}
