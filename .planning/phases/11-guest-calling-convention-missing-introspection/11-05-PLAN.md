```rust
/// Host function: list all loaded bundles.
pub fn host_list_bundles(host: *const HostInterface) -> Array<BundleId> {
    // Extract runtime from interface
    // SAFETY: host pointer validated by caller
    let host_ref = unsafe { &*host };
    let runtime = host_ref.runtime as *const Runtime;
    // SAFETY: runtime pointer is valid for lifetime of interface
    let runtime_ref = unsafe { &*runtime };
    
    let manifests = runtime_ref.bundle_manifests.lock().unwrap();
    let count = manifests.len();
    
    if count == 0 {
        return Array { items: core::ptr::null_mut(), len: 0, align: 0 };
    }
    
    // Allocate via host allocator
    let size = count * core::mem::size_of::<BundleId>();
    let align = core::mem::align_of::<BundleId>();
    let ptr = host_alloc(host, size, align) as *mut BundleId;
    
    if ptr.is_null() {
        return Array { items: core::ptr::null_mut(), len: 0, align: 0 };
    }
    
    // Fill array
    for (i, (_, manifest)) in manifests.iter().enumerate() {
        unsafe {
            *ptr.add(i) = BundleId::from_u64(manifest.id);
        }
    }
    
    Array { items: ptr, len: count, align }
}
```
