       pub fn get_host_contract(
           &self,
           contract_id: u64,
           min_version: u32,
       ) -> Option<HostContractInstance> {
           let interface: &HostContractInterface = self.host_contracts.read()
               .unwrap_or_else(|e| e.into_inner())
               .values()
               .find(|iface| {
                   iface.contract_id.0 == contract_id &&
                   iface.contract_version.major >= (min_version >> 16)
               })?;
           
           if interface.singleton {
               // Check cache first
               let cache: RwLockReadGuard<HashMap<u64, HostContractInstance>> = 
                   self.singleton_instances.read().unwrap_or_else(|e| e.into_inner());
               if let Some(instance) = cache.get(&contract_id) {
                   return Some(*instance);
               }
               // Need to create and cache
               drop(cache);

                   self.singleton_instances.write().unwrap_or_else(|e| e.into_inner());

               if let Some(instance) = cache.get(&contract_id) {

               Some(instance)

           rt_ctx: *mut core::ffi::c_void,
           contract_id: u64,
           min_version: u32,
       ) -> polyplug_abi::HostContractInstance {
           if rt_ctx.is_null() {
               return polyplug_abi::HostContractInstance { data: core::ptr::null_mut() };
           }
           // SAFETY: rt_ctx is a valid *mut HostContext
           let ctx: &HostContext = unsafe { &*(rt_ctx as *const HostContext) };
           // SAFETY: ctx.runtime is a valid pointer to a live Runtime
           let runtime: &Runtime = unsafe { &*(ctx.runtime as *const Runtime) };
       
           // Find the host contract interface
           let host_contracts: RwLockReadGuard<HashMap<u64, &'static HostContractInterface>> =
               runtime.host_contracts.read().unwrap_or_else(|e| e.into_inner());
       
           let interface: &HostContractInterface = host_contracts.values()
               .find(|iface| {
                   iface.contract_id.0 == contract_id &&
                   iface.contract_version.major >= (min_version >> 16)
               })?;
       
           if interface.singleton {
               // Check singleton cache
               let singleton_instances: RwLockReadGuard<HashMap<u64, HostContractInstance>> =
                   runtime.singleton_instances.read().unwrap_or_else(|e| e.into_inner());
               if let Some(instance) = singleton_instances.get(&contract_id) {
                   return *instance;
               }
               drop(singleton_instances);
       
               // Create singleton and cache it
               let mut singleton_instances: RwLockWriteGuard<HashMap<u64, HostContractInstance>> =
                   runtime.singleton_instances.write().unwrap_or_else(|e| e.into_inner());
               // Double-check pattern
               if let Some(instance) = singleton_instances.get(&contract_id) {
                   return *instance;
               }
               let instance: HostContractInstance = unsafe {
                   (interface.create_instance)(rt_ctx, core::ptr::null())
               };
               singleton_instances.insert(contract_id, instance);
               instance
           } else {
               // Multi-instance: create new instance
               unsafe {
                   (interface.create_instance)(rt_ctx, core::ptr::null())
               }
           }
       }
