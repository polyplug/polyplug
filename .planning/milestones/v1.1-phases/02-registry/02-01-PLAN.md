<objective>
Remove VTableSlot wrapper and PluginGuard RAII pattern, replacing with direct Arc<GuestContractInterface> storage in RegistrySlot.

Purpose: The VTableSlot wrapper and PluginGuard RAII pattern were designed for Arc-based quiescence tracking in hot-reload. The new instance-based model (Phase 3) makes these unnecessary because hosts explicitly destroy instances before hot-reload via callback.

Output: Registry stores interface directly, resolve() returns interface pointer without guard.
</objective>

        ///
        /// This is a direct swap under write lock. The callback-based hot-reload model
        /// (Phase 4) ensures hosts destroy instances before this is called.
        pub fn swap_interface(
            &self,
            slot_index: u32,
            new_interface: Arc<GuestContractInterface>,
        ) -> Result<(), RegistryError> {
            let mut data = self.data.write().unwrap_or_else(|e| {
                eprintln!("[polyplug] Mutex/RwLock poisoned, recovering: {}", e);
                e.into_inner()
            });
            let slot_idx = slot_index as usize;
            if slot_idx >= data.slots.len() {
                return Err(RegistryError::StaleHandle {
                    index: slot_index,
                    expected: 0,
                    found: 0,
                });
            }
            let slot = &mut data.slots[slot_idx];
            if slot.interface.is_none() {
                return Err(RegistryError::StaleHandle {
                    index: slot_index,
                    expected: 0,
                    found: slot.generation.load(Ordering::Acquire),
                });
            }
            slot.interface = Some(new_interface);
            // Bump generation for stale handle detection (removed in Plan 03)
            slot.generation.fetch_add(1, Ordering::AcqRel);
            Ok(())
        }

       pub fn resolve(&self, handle: GuestContractHandle) -> Result<*const GuestContractInterface, RegistryError> {
           let data = self.data.read().unwrap_or_else(|e| {
               eprintln!("[polyplug] Mutex/RwLock poisoned, recovering: {}", e);
               e.into_inner()
           });

           let slot_idx = handle.index as usize;
           if slot_idx >= data.slots.len() {
               return Err(RegistryError::StaleHandle {
                   index: handle.index,
                   expected: handle.generation,
                   found: 0,
               });
           }

           let slot = &data.slots[slot_idx];
           let slot_generation = slot.generation.load(Ordering::Acquire);
           if slot_generation != handle.generation {
               return Err(RegistryError::StaleHandle {
                   index: handle.index,
                   expected: handle.generation,
                   found: slot_generation,
               });

               Some(ref arc) => Ok(arc.as_ref() as *const GuestContractInterface),
               None => Err(RegistryError::StaleHandle {
                   index: handle.index,
                   expected: handle.generation,
                   found: slot_generation,
               }),
